1. Instanciation d'un Héros en AnimatedSprite

import { Assets, AnimatedSprite } from 'pixi.js';

// Charger l'animation de marche du Minotaure
const framesMarche = [];
for (let i = 1; i <= 6; i++) {
  const num = String(i).padStart(2, '0');
  const tex = await Assets.load(`dossier_claude/personnages/minotaure/frames_animation/minotaure_marche_${num}.webp`);
  framesMarche.push(tex);
}

const minotaureMarche = new AnimatedSprite(framesMarche);
minotaureMarche.anchor.set(0.5, 0.95);
minotaureMarche.animationSpeed = 0.15; // ~9 FPS
minotaureMarche.play();
app.stage.addChild(minotaureMarche);

2. Amélioration Dynamique de Bâtiment (Niveau 1 à 5)

import { Assets, Sprite } from 'pixi.js';

async function mettreAJourBatiment(spriteExistant, type, nouveauNiveau) {
  const nouvelleTexture = await Assets.load(`dossier_claude/batiments/${type}/sprites_decoupes/${type}_niveau_${nouveauNiveau}.webp`);
  spriteExistant.texture = nouvelleTexture;
  
  // Effet d'échelle doux pour signaler la montée de niveau
  spriteExistant.scale.set(1.1);
  setTimeout(() => spriteExistant.scale.set(1.0), 300);
}

3. Évaluation d'IA en Combat (Soda Scripts)

function executerTourHeros(heros, ciblesEnemies, allies) {
  // Règle 1 : Soin d'urgence si allié bas en PV
  if (heros.classe === 'Satyre') {
    const allieEnDanger = allies.find(a => a.pv / a.pvMax < 0.5);
    if (allieEnDanger) return { action: 'soin', cible: allieEnDanger };
  }
  
  // Règle 2 : Frappe lourde du Minotaure sur le Boss ou la cible la plus dangereuse
  if (heros.classe === 'Minotaure') {
    const boss = ciblesEnemies.find(e => e.estBoss);
    if (boss) return { action: 'charge_tellurique', cible: boss };
  }
  
  // Règle par défaut
  return { action: 'attaque_simple', cible: ciblesEnemies[0] };
}
